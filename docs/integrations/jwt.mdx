# JWT Authentication

ExisJS comes with a native JSON Web Token (JWT) integration. You do **not** need to install external dependencies like `jsonwebtoken`. It is built natively into the framework and optimized for edge environments.

---

## Environment Variables

The JWT integration securely reads your secret key from your `.env` file. 
Before using the plugin, ensure you have your JWT secret set:

```bash filename=".env"
# Your JWT signing secret
JWT_SECRET="super_secret_key_123"
```

---

## Usage

Import the `JWT` object from `exisjs/auth` and use `JWT.signJWT()` or `JWT.verifyJWT()`. You will need to explicitly pass your secret key.

#### Functional Paradigm

```typescript filename="src/http/auth/route.ts"
import { controller, route } from 'exisjs/router'
import { JWT } from 'exisjs/auth'
import { db, users } from '../../db'

export default controller({
  login: route.post('/login', {
    async handle(req, res) {
      const user = await db.select().from(users).where({ email: req.body.email })
      
      if (!user) {
        res.status(401)
        return { error: 'Invalid credentials' }
      }

      // Sign the token using process.env.JWT_SECRET
      const token = JWT.signJWT({ userId: user.id }, process.env.JWT_SECRET!, { expiresIn: 604800 })
      
      return { token }
    }
  }),

  verify: route.get('/me', {
    async handle(req, res) {
      try {
        const token = req.headers.authorization?.split(' ')[1]
        
        // Verify using process.env.JWT_SECRET
        const decoded = JWT.verifyJWT<{ userId: string }>(token, process.env.JWT_SECRET!)
        
        return { userId: decoded.userId }
      } catch (err) {
        res.status(401)
        return { error: 'Invalid or expired token' }
      }
    }
  })
})
```

#### Class-Based (OOP) Paradigm

```typescript filename="src/http/auth/route.ts"
import { Controller, Post, Get, Body, Req, Res } from 'exisjs/decorators'
import { JWT } from 'exisjs/auth'
import { db, users } from '../../db'

@Controller('/auth')
export default class AuthController {
  @Post('/login')
  async login(@Body() body: any, @Res() res: any) {
    const user = await db.select().from(users).where({ email: body.email })
    
    if (!user) {
      res.status(401)
      return { error: 'Invalid credentials' }
    }

    // Sign the token using process.env.JWT_SECRET
    const token = JWT.signJWT({ userId: user.id }, process.env.JWT_SECRET!, { expiresIn: 604800 })
    
    return { token }
  }

  @Get('/me')
  async verifyMe(@Req() req: any, @Res() res: any) {
    try {
      const token = req.headers.authorization?.split(' ')[1]
      
      // Verify using process.env.JWT_SECRET
      const decoded = JWT.verifyJWT<{ userId: string }>(token, process.env.JWT_SECRET!)
      
      return { userId: decoded.userId }
    } catch (err) {
      res.status(401)
      return { error: 'Invalid or expired token' }
    }
  }
}
```

## Using JWT in Middleware & Gateways

While you can verify tokens directly in your routes, ExisJS's JWT integration works well inside Middlewares to protect entire folders using **Gateways**.

First, create your middleware outside of your routing tree:

```typescript filename="src/middlewares/auth.ts"
import { JWT } from 'exisjs/auth'

export function requireAuth(req: any, res: any, next: any) {
  const token = req.headers.authorization?.split(' ')[1]
  
  if (!token) {
    res.status(401).json({ error: 'Missing token' })
    return
  }

  try {
    const decoded = JWT.verifyJWT<{ userId: string }>(token, process.env.JWT_SECRET!)
    // Attach the decoded payload to the request for the route to use
    req.user = decoded
    next()
  } catch (err) {
    res.status(401).json({ error: 'Unauthorized' })
  }
}
```

Now, apply it to an entire domain using a Gateway. This prevents you from having to protect every route manually.

```typescript filename="src/http/api/gateway.ts"
import { defineGateway } from 'exisjs/router'
import { requireAuth } from '../../middlewares/auth'

export default defineGateway({
  // Intercepts and protects every route inside the /api folder
  middleware: [requireAuth]
})
```
